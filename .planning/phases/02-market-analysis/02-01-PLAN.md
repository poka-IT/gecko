    this.sentCount = 0,
    this.receivedCount = 0,
  });

  int get transactionCount => sentCount + receivedCount;
}

**MarketAnalysisService class methods:**

1. `Future<List<TransactionDisplayItem>> fetchAllPages(String walletAddress, String contactAddress, DateTime startDate, DateTime endDate, DateTime genesisTime)`:
   - Creates `TransactionFilters(addresses: [contactAddress], exactMatchAddress: true, startDate: startDate.toUtc(), endDate: endDate.toUtc())`
   - Loops calling `SquidService.client.getAccountHistoryFiltered(walletAddress, number: 50, cursor: cursor, filters: filters)` until `hasNextPage` is false
   - Converts each node to `TransactionDisplayItem.fromFilteredGraphQLNode(node, walletAddress, genesisTime)`
   - Returns accumulated list of all items
   - If result is null (Squid offline), returns empty list

2. `ContactAnalysisResult aggregateTransactions(List<TransactionDisplayItem> items, String contactAddress)`:
   - Iterates items, accumulates `totalSent` (where `isReceived == false`), `totalReceived` (where `isReceived == true`), `sentCount`, `receivedCount`
   - Gets username from the first item with a non-null `username`, or null
   - Returns `ContactAnalysisResult` with the totals

3. `Map<String, ContactAnalysisResult> discoverOtherContacts(List<TransactionDisplayItem> allItems, String walletAddress, Set<String> selectedAddresses)`:
   - Collects all unique `address` fields from allItems (the "other party" address)
   - Filters out: walletAddress itself, all addresses in selectedAddresses, empty strings
   - For each remaining address, aggregates its items using `aggregateTransactions`
   - Returns map of address -> ContactAnalysisResult

4. `String generateMarkdownReport(...)` with parameters: `String walletName`, `DateTime startDate`, `DateTime endDate`, `Map<String, ContactAnalysisResult> contactResults`, `Map<String, ContactAnalysisResult> otherContactResults`:
   - Uses `DateFormat('yyyy-MM-dd')` from `intl` package for date formatting
   - Generates markdown per D-12:
     ```
     # Market Analysis
     **Wallet:** {walletName}
     **Period:** {startDate} - {endDate}

     ## Summary
     | Contact | Sent | Received | Transactions |
     |---------|------|----------|-------------|
     | {name or shortPubkey} | {sent/100 with 2 decimals} | {received/100 with 2 decimals} | {count} |
     | **Total** | **{totalSent}** | **{totalReceived}** | **{totalCount}** |

     ## Other Contacts
     | Contact | Sent | Received | Transactions |
     |---------|------|----------|-------------|
     (same format for otherContactResults, only if non-empty)
     ```
   - Uses `getShortPubkey(address)` from `utils.dart` when username is null
   - Amount conversion: `(amount.toDouble() / 100).toStringAsFixed(2)` for markdown text only (NOT for UI display)

**MarketAnalysisState class:**
```dart
class MarketAnalysisState {
  final DateTime? startDate;
  final DateTime? endDate;
  final Set<String> selectedContactAddresses;
  final Map<String, ContactAnalysisResult> contactResults;
  final Map<String, ContactAnalysisResult> otherContactResults;
  final bool isAnalyzing;
  final int processedContacts;
  final int totalContacts;
  final String? error;

  const MarketAnalysisState({
    this.startDate,
    this.endDate,
    this.selectedContactAddresses = const {},
    this.contactResults = const {},
    this.otherContactResults = const {},
    this.isAnalyzing = false,
    this.processedContacts = 0,
    this.totalContacts = 0,
    this.error,
  });

  MarketAnalysisState copyWith({...}); // all fields

  BigInt get totalSent => contactResults.values.fold(BigInt.zero, (sum, r) => sum + r.totalSent);
  BigInt get totalReceived => contactResults.values.fold(BigInt.zero, (sum, r) => sum + r.totalReceived);
  int get totalTransactionCount => contactResults.values.fold(0, (sum, r) => sum + r.transactionCount);

  bool get canAnalyze => startDate != null && endDate != null && selectedContactAddresses.isNotEmpty;

  bool get hasResults => contactResults.isNotEmpty;
}
```

**MarketAnalysisNotifier** extends `Notifier<MarketAnalysisState>`:

Methods:
- `void setDateRange(DateTime start, DateTime end)` -- validates max 365 days (per D-02), sets state or sets error if exceeded
- `void toggleContact(String address)` -- adds/removes from selectedContactAddresses set
- `void selectAllContacts(List<String> addresses)` -- replaces selectedContactAddresses with all (per D-04)
- `void deselectAllContacts()` -- clears selectedContactAddresses (per D-04)
- `Future<void> runAnalysis(String walletAddress)` -- the main analysis flow:
  1. Check `ref.read(squidConnectionStatusProvider)` is `ConnectionStatus.connected`. If not, set `error: 'noNetworkConnection'` and return.
  2. Set `isAnalyzing: true`, clear previous results, set `totalContacts` and `processedContacts: 0`
  3. Read `genesisTimeProvider.future` for genesis time
  4. Read `marketAnalysisServiceProvider` for the service
  5. Loop through `selectedContactAddresses` sequentially:
     - Call `service.fetchAllPages(walletAddress, contactAddress, startDate!, endDate!, genesisTime)`
     - Call `service.aggregateTransactions(items, contactAddress)`
     - Resolve username via `ref.read(squidServiceProvider).walletNameIndexer[contactAddress]` if result.username is null
     - Update state progressively: add to contactResults, increment processedContacts
     - Collect all items for other-contacts discovery
  6. After loop, call `service.discoverOtherContacts(allItems, walletAddress, selectedContactAddresses)`
  7. For each other-contact, try resolving name via `squidServiceProvider.walletNameIndexer[address]`
  8. Set `otherContactResults` and `isAnalyzing: false`
  9. Wrap entire flow in try/catch, set `error` on failure and `isAnalyzing: false`
- `void reset()` -- returns to initial state
